class AdminUserViewSet(
    mixins.ListModelMixin, mixins.RetrieveModelMixin, viewsets.GenericViewSet
):
    queryset = User.objects.all()
    serializer_class = AdminUserSerializer
    permission_classes = [IsAdministrator]
    filterset_fields = ["role", "is_blocked", "is_active"]
    search_fields = ["username", "email", "phone", "first_name", "last_name"]
    ordering_fields = ["date_joined", "username", "last_login"]
    ordering = ["-date_joined"]

    def _respond(self, user):
        return Response(self.get_serializer(user).data)

    @action(detail=True, methods=["post"])
    def block(self, request, pk=None):
        user = self.get_object()
        data = BlockSerializer(data=request.data)
        data.is_valid(raise_exception=True)

        if user.pk == request.user.pk:
            raise ValidationError("O'zingizni bloklay olmaysiz.")
        if user.is_administrator:
            raise ValidationError("Administratorni bloklab bo'lmaydi.")
        if user.is_blocked:
            raise ValidationError("Foydalanuvchi allaqachon bloklangan.")

        user.is_blocked = True
        user.blocked_reason = data.validated_data["reason"][:255]
        user.save(update_fields=["is_blocked", "blocked_reason"])
        Token.objects.filter(user=user).delete()  # darhol tizimdan chiqadi
        return self._respond(user)

    @action(detail=True, methods=["post"])
    def unblock(self, request, pk=None):
        user = self.get_object()
        if not user.is_blocked:
            raise ValidationError("Foydalanuvchi bloklanmagan.")
        user.is_blocked = False
        user.blocked_reason = ""
        user.save(update_fields=["is_blocked", "blocked_reason"])
        return self._respond(user)

    @action(detail=True, methods=["post"], url_path="set-role")
    def set_role(self, request, pk=None):
        user = self.get_object()
        data = RoleSerializer(data=request.data)
        data.is_valid(raise_exception=True)
        new_role = data.validated_data["role"]

        if user.pk == request.user.pk:
            raise ValidationError("O'z rolingizni o'zgartira olmaysiz.")
        if user.is_administrator:
            raise ValidationError("Administrator rolini API orqali o'zgartirib bo'lmaydi.")
        if user.role == new_role:
            raise ValidationError("Foydalanuvchi allaqachon shu rolda.")

        user.role = new_role
        user.save()  # is_staff ni save() o'zi moslaydi
        label = user.get_role_display()
        notify(
            user,
            Notification.Kind.SYSTEM,
            "Rolingiz o'zgardi",
            f"Sizning yangi rolingiz: {label}.",
        )
        return self._respond(user)

from django.contrib.auth.models import AbstractUser
from django.db import models


class User(AbstractUser):
    class Role(models.TextChoices):
        USER = "user", "Foydalanuvchi"
        MODERATOR = "moderator", "Moderator"
        ADMIN = "admin", "Administrator"

    phone = models.CharField("Telefon", max_length=20, blank=True)
    role = models.CharField("Rol", max_length=20, choices=Role.choices, default=Role.USER)
    is_blocked = models.BooleanField("Bloklangan", default=False)
    blocked_reason = models.CharField("Bloklash sababi", max_length=255, blank=True)

    class Meta:
        verbose_name = "Foydalanuvchi"
        verbose_name_plural = "Foydalanuvchilar"

    def save(self, *args, **kwargs):
        # Superuser avtomatik admin bo'ladi, Django admin paneliga faqat adminlar kiradi
        if self.is_superuser:
            self.role = self.Role.ADMIN
        self.is_staff = self.is_superuser or self.role == self.Role.ADMIN
        super().save(*args, **kwargs)

    @property
    def is_administrator(self):
        return self.role == self.Role.ADMIN

    @property
    def is_moderator(self):
        """Moderator ishini moderator ham, admin ham qila oladi."""
        return self.role in (self.Role.MODERATOR, self.Role.ADMIN)

    def __str__(self):
        return self.username

    def has_permission(self, request, view):
        user = request.user
        return bool(user and user.is_authenticated and user.is_administrator)

    """Hamma ko'ra oladi, o'zgartirish faqat admin."""

    def has_permission(self, request, view):
        if request.method in SAFE_METHODS:
            return True
        user = request.user
        return bool(user and user.is_authenticated and user.is_administrator)

    def validate_password(self, value):
        password_validation.validate_password(value)
        return value

    def create(self, validated_data):
        # rol har doim "user", ro'yxatdan o'tishda o'zgartirib bo'lmaydi
        return User.objects.create_user(**validated_data)


class LoginSerializer(serializers.Serializer):
    username = serializers.CharField()
    password = serializers.CharField(write_only=True)

    def validate(self, attrs):
        user = authenticate(username=attrs["username"], password=attrs["password"])
        if not user:
            raise serializers.ValidationError("Login yoki parol noto'g'ri.")
        if user.is_blocked:
            reason = f" Sabab: {user.blocked_reason}" if user.blocked_reason else ""
            raise serializers.ValidationError("Akkaunt bloklangan." + reason)
        attrs["user"] = user
        return attrs


class UserSerializer(serializers.ModelSerializer):
    class Meta:
        model = User
        fields = (
            "id", "username", "email", "phone", "first_name", "last_name",
            "role", "is_blocked", "date_joined",
        )
        read_only_fields = ("id", "username", "role", "is_blocked", "date_joined")


class ChangePasswordSerializer(serializers.Serializer):
    old_password = serializers.CharField(write_only=True)
    new_password = serializers.CharField(write_only=True, min_length=6)

    def validate_old_password(self, value):
        if not self.context["request"].user.check_password(value):
            raise serializers.ValidationError("Eski parol noto'g'ri.")
        return value

    def validate_new_password(self, value):
        password_validation.validate_password(value, self.context["request"].user)
        return value

from rest_framework import status
from rest_framework.authtoken.models import Token
from rest_framework.decorators import api_view, permission_classes
from rest_framework.permissions import AllowAny, IsAuthenticated
from rest_framework.response import Response

from .serializers import (
    ChangePasswordSerializer,
    LoginSerializer,
    RegisterSerializer,
    UserSerializer,
)


@api_view(["POST"])
@permission_classes([AllowAny])
def register(request):
    serializer = RegisterSerializer(data=request.data)
    serializer.is_valid(raise_exception=True)
    user = serializer.save()
    token, _ = Token.objects.get_or_create(user=user)
    return Response(
        {"token": token.key, "user": UserSerializer(user).data},
        status=status.HTTP_201_CREATED,
    )


@api_view(["POST"])
@permission_classes([AllowAny])
def login(request):
    serializer = LoginSerializer(data=request.data)
    serializer.is_valid(raise_exception=True)
    user = serializer.validated_data["user"]
    token, _ = Token.objects.get_or_create(user=user)
    return Response({"token": token.key, "user": UserSerializer(user).data})


@api_view(["POST"])
@permission_classes([IsAuthenticated])
def logout(request):
    Token.objects.filter(user=request.user).delete()
    return Response(status=status.HTTP_204_NO_CONTENT)


@api_view(["GET", "PATCH"])
@permission_classes([IsAuthenticated])
def me(request):
    if request.method == "PATCH":
        serializer = UserSerializer(request.user, data=request.data, partial=True)
        serializer.is_valid(raise_exception=True)
        serializer.save()
    return Response(UserSerializer(request.user).data)


@api_view(["POST"])
@permission_classes([IsAuthenticated])
def change_password(request):
    serializer = ChangePasswordSerializer(data=request.data, context={"request": request})
    serializer.is_valid(raise_exception=True)
    request.user.set_password(serializer.validated_data["new_password"])
    request.user.save()
    # eski tokenlar bekor, yangisi beriladi
    Token.objects.filter(user=request.user).delete()
    token = Token.objects.create(user=request.user)
    return Response({"token": token.key})

            "description", "proof_image", "status", "status_display",

        )
        read_only_fields = (
            "status", "moderator_note", "handover_note", "handed_over_at", "created_at",
        )

    def get_lost_post_title(self, obj):
        return obj.lost_post.title if obj.lost_post else None

    def get_pickup_info(self, obj):
        # Buyum qayerdaligi faqat da'vo tasdiqlangandan keyin ko'rinadi
        if obj.status in (Claim.Status.APPROVED, Claim.Status.COMPLETED):
            return obj.found_post.storage_note
        return None

    def validate_description(self, value):
        value = value.strip()
        if len(value) < 10:
            raise serializers.ValidationError(
                "Buyumni tasdiqlovchi belgilarni batafsilroq yozing (kamida 10 belgi)."
            )
        return value

    def validate_found_post(self, post):
        if post.type != Post.Type.FOUND or post.status != Post.Status.ACTIVE:
            raise serializers.ValidationError("Bu e'longa da'vo yuborib bo'lmaydi.")
        return post

    def validate(self, attrs):
        user = self.context["request"].user
        found = attrs["found_post"]

        if found.author_id == user.id:
            raise serializers.ValidationError("O'z e'loningizga da'vo yubora olmaysiz.")

        lost = attrs.get("lost_post")
        if lost and (
            lost.type != Post.Type.LOST
            or lost.author_id != user.id
            or lost.status != Post.Status.ACTIVE
        ):
            raise serializers.ValidationError(
                {"lost_post": "Faqat o'zingizning faol yo'qolgan e'loningizni tanlang."}
            )

        if Claim.objects.filter(
            found_post=found, claimant=user, status=Claim.Status.PENDING
        ).exists():
            raise serializers.ValidationError(
                "Bu buyum uchun da'voyingiz allaqachon ko'rib chiqilmoqda."
            )
        return attrs

def _require_reason(reason):
    reason = (reason or "").strip()

@transaction.atomic
def approve_claim(claim, moderator, note=""):
    if claim.status != Claim.Status.PENDING:
        raise ValidationError("Faqat ko'rib chiqilayotgan da'voni tasdiqlash mumkin.")
    if claim.claimant_id == moderator.id:
        raise ValidationError("O'z da'vongizni o'zingiz tasdiqlay olmaysiz.")

    found = claim.found_post
    if found.status != Post.Status.ACTIVE:
        raise ValidationError("Topilgan e'lon faol emas.")

    now = timezone.now()
    claim.status = Claim.Status.APPROVED
    claim.moderator = moderator
    claim.moderator_note = (note or "")[:255]
    claim.reviewed_at = now
    claim.save()

    found.status = Post.Status.HANDOVER
    found.save(update_fields=["status", "updated_at"])

    lost = claim.lost_post
    if lost and lost.status == Post.Status.ACTIVE:
        lost.status = Post.Status.HANDOVER
        lost.save(update_fields=["status", "updated_at"])

    # Shu buyumga boshqa kutilayotgan da'volar bekor qilinadi
    others = Claim.objects.filter(found_post=found, status=Claim.Status.PENDING).exclude(
        pk=claim.pk
    ).select_related("claimant")
    for other in others:
        other.status = Claim.Status.CANCELLED
        other.moderator = moderator
        other.moderator_note = "Bu buyum uchun boshqa da'vo tasdiqlandi."
        other.reviewed_at = now
        other.save()
        notify(
            other.claimant,
            Notification.Kind.CLAIM_REJECTED,
            "Da'vo bekor qilindi",
            other.moderator_note,
            post=found,
            claim=other,
        )

    pickup = found.storage_note or "Moderator siz bilan bog'lanadi."
    notify(
        claim.claimant,
        Notification.Kind.CLAIM_APPROVED,
        "Da'voyingiz tasdiqlandi",
        f"«{found.title}» egasi sifatida tasdiqlandingiz. Buyumni olish: {pickup}",
        post=found,
        claim=claim,
    )
    notify(
        found.author,
        Notification.Kind.CLAIM_APPROVED,
        "Buyum egasi aniqlandi",
        f"«{found.title}» uchun egasi tasdiqlandi. Topshirish jarayoni boshlandi.",
        post=found,
    )
    return claim


@transaction.atomic
def reject_claim(claim, moderator, reason):
    if claim.status != Claim.Status.PENDING:
        raise ValidationError("Faqat ko'rib chiqilayotgan da'voni rad etish mumkin.")
    reason = _require_reason(reason)

    claim.status = Claim.Status.REJECTED
    claim.moderator = moderator
    claim.moderator_note = reason
    claim.reviewed_at = timezone.now()
    claim.save()

    notify(
        claim.claimant,
        Notification.Kind.CLAIM_REJECTED,
        "Da'voyingiz rad etildi",
        f"«{claim.found_post.title}»: {reason}",
        post=claim.found_post,
        claim=claim,
    )
    return claim


@transaction.atomic
def complete_claim(claim, moderator, note=""):
    if claim.status != Claim.Status.APPROVED:
        raise ValidationError("Faqat tasdiqlangan da'voni topshirilgan deb belgilash mumkin.")

    now = timezone.now()
    claim.status = Claim.Status.COMPLETED
    claim.handover_note = (note or "")[:255]
    claim.handed_over_at = now
    claim.save()

    for post in (claim.found_post, claim.lost_post):
        if post and post.status == Post.Status.HANDOVER:
            post.status = Post.Status.CLOSED
            post.closed_at = now
            post.save(update_fields=["status", "closed_at", "updated_at"])

    found = claim.found_post
    notify(
        claim.claimant,
        Notification.Kind.HANDOVER_COMPLETED,
        "Buyum topshirildi",
        f"«{found.title}» sizga topshirildi. Yaxshi kun tilaymiz!",
        post=found,
        claim=claim,
    )
    notify(
        found.author,
        Notification.Kind.HANDOVER_COMPLETED,
        "Buyum egasiga qaytarildi",
        f"«{found.title}» egasiga topshirildi. Yordamingiz uchun rahmat!",
        post=found,
    )
    return claim


@transaction.atomic
def revoke_claim(claim, moderator, reason):
    """Tasdiqlangan, lekin hali topshirilmagan da'voni bekor qilish."""
    if claim.status != Claim.Status.APPROVED:

    reason = _require_reason(reason)

    claim.status = Claim.Status.CANCELLED
    claim.moderator = moderator
    claim.moderator_note = reason
    claim.save()

    for post in (claim.found_post, claim.lost_post):
        if post and post.status == Post.Status.HANDOVER:

        claim.claimant,
        Notification.Kind.CLAIM_REJECTED,
        "Tasdiq bekor qilindi",

    )

        claim.found_post.author,
        Notification.Kind.SYSTEM,
        "Topshirish bekor qilindi",
        f"«{claim.found_post.title}» e'loni yana faol.",
        post=claim.found_post,
    )
    return claim

class ModerationClaimViewSet(
    mixins.ListModelMixin, mixins.RetrieveModelMixin, viewsets.GenericViewSet
):
    serializer_class = ModerationClaimSerializer
    permission_classes = [IsModerator]
    filterset_fields = ["status"]
    ordering_fields = ["created_at"]
    ordering = ["created_at"]  # navbat: eskisi birinchi

    def get_queryset(self):
        qs = Claim.objects.select_related(
            "claimant", "moderator",

@api_view(["GET", "PATCH"])

        serializer = SystemSettingsSerializer(obj, data=request.data, partial=True)

def _counts(qs, field, choices):
    """Har bir variant bo'yicha soni (0 bo'lsa ham ko'rsatiladi)."""
    data = dict(qs.values_list(field).annotate(c=Count("id")).order_by())
    return {key: data.get(key, 0) for key, _ in choices}


@api_view(["GET"])
@permission_classes([IsAdministrator])
def statistics(request):
    now = timezone.now()
    week_ago = now - timedelta(days=7)

    posts = Post.objects.all()
    claims = Claim.objects.all()
    reports = Report.objects.all()
    users = User.objects.all()

    found_total = posts.filter(
        type=Post.Type.FOUND,
        status__in=[Post.Status.ACTIVE, Post.Status.HANDOVER, Post.Status.CLOSED],
    ).count()
    returned = claims.filter(status=Claim.Status.COMPLETED).count()

    # Oxirgi 14 kun: kuniga nechta yo'qolgan/topilgan e'lon
    days = 14
    today = timezone.localdate()
    start_day = today - timedelta(days=days - 1)
    rows = (
        posts.filter(created_at__date__gte=start_day)
        .annotate(day=TruncDate("created_at"))
        .values("day", "type")
        .annotate(c=Count("id"))
        .order_by()
    )
    grid = {}
    for r in rows:
        grid.setdefault(r["day"], {})[r["type"]] = r["c"]
    trend = []
    for i in range(days):
        d = start_day + timedelta(days=i)
        trend.append({
            "date": d.isoformat(),
            "lost": grid.get(d, {}).get(Post.Type.LOST, 0),
            "found": grid.get(d, {}).get(Post.Type.FOUND, 0),
        })

    top_categories = [
        {"name": r["category__name"], "count": r["c"]}
        for r in posts.values("category__name").annotate(c=Count("id")).order_by("-c")[:10]
    ]

    return Response({
        "users": {
            "total": users.count(),
            "by_role": _counts(users, "role", User.Role.choices),
            "blocked": users.filter(is_blocked=True).count(),
            "new_last_7_days": users.filter(date_joined__gte=week_ago).count(),
        },
        "posts": {
            "total": posts.count(),
            "by_status": _counts(posts, "status", Post.Status.choices),
            "by_type": _counts(posts, "type", Post.Type.choices),
            "active_by_type": _counts(
                posts.filter(status=Post.Status.ACTIVE), "type", Post.Type.choices
            ),
            "per_day": trend,
            "top_categories": top_categories,
        },
        "claims": {
            "total": claims.count(),
            "by_status": _counts(claims, "status", Claim.Status.choices),
        },
        "reports": {
            "total": reports.count(),
            "by_status": _counts(reports, "status", Report.Status.choices),
        },
        "matches": {"total": Match.objects.count()},
        "returns": {
            "returned_items": returned,
            "found_items_total": found_total,
            "return_rate_percent": round(returned / found_total * 100, 1) if found_total else 0,
        },
        "queues": {
            "pending_posts": posts.filter(status=Post.Status.PENDING).count(),
            "pending_claims": claims.filter(status=Claim.Status.PENDING).count(),
            "open_reports": reports.filter(status=Report.Status.OPEN).count(),
        },
    })

import math
import re

from django.db.models import Q

from core.models import SystemSettings
from notifications.models import Notification
from notifications.services import notify

from .models import Match, Post

WEIGHTS = {"color": 20, "brand": 15, "location": 20, "date": 15, "text": 30}


def haversine_km(lat1, lon1, lat2, lon2):
    """Ikki nuqta orasidagi masofa (km)."""
    r = 6371.0
    p1, p2 = math.radians(lat1), math.radians(lat2)
    dphi = p2 - p1
    dlmb = math.radians(lon2 - lon1)
    a = math.sin(dphi / 2) ** 2 + math.cos(p1) * math.cos(p2) * math.sin(dlmb / 2) ** 2
    return 2 * r * math.asin(math.sqrt(a))


def tokens(text):
    text = re.sub(r"[ʻʼ’‘'`]", "", (text or "").lower())
    return {w for w in re.findall(r"\w+", text) if len(w) >= 3}


def color_points(a, b):
    other = Post.Color.OTHER
    if a == b:
        return WEIGHTS["color"] if a != other else WEIGHTS["color"] // 2
    if other in (a, b):
        return 8
    return 0


def brand_points(a, b):
    a = re.sub(r"\W+", "", (a or "").lower())
    b = re.sub(r"\W+", "", (b or "").lower())
    if a and b and (a == b or a in b or b in a):
        return WEIGHTS["brand"]
    return 0


def location_points(lost, found, config):
    """(ball, masofa) yoki juda uzoq bo'lsa None."""
    distance = None
    if lost.has_coordinates and found.has_coordinates:
        distance = haversine_km(lost.latitude, lost.longitude, found.latitude, found.longitude)
        if distance > config.match_max_distance_km:
            return None

    if lost.place_id and lost.place_id == found.place_id:
        return WEIGHTS["location"], distance
    if distance is None:
        return 0, None
    if distance <= 0.3:
        return 20, distance
    if distance <= 1:
        return 15, distance
    if distance <= 3:
        return 10, distance
    return 5, distance


def date_points(lost, found, config):
    """Sana bo'yicha ball yoki mantiqsiz bo'lsa None."""
    diff = (found.event_date - lost.event_date).days
    if diff < -1 or diff > config.match_date_window_days:
        return None
    if diff <= 0:
        return 15
    if diff <= 1:
        return 12
    if diff <= 3:
        return 9
    if diff <= 7:
        return 6
    return 3


def text_points(lost, found):
    a = tokens(f"{lost.title} {lost.description}")
    b = tokens(f"{found.title} {found.description}")
    if not a or not b:
        return 0
    overlap = len(a & b) / min(len(a), len(b))
    return round(WEIGHTS["text"] * overlap)


def compute_score(lost, found, config):
    """(ball, tafsilot) yoki juft mos kelmasa None."""
    loc = location_points(lost, found, config)
    if loc is None:
        return None
    date_pts = date_points(lost, found, config)
    if date_pts is None:
        return None

    location_pts, distance = loc
    reasons = {
        "color": color_points(lost.color, found.color),
        "brand": brand_points(lost.brand, found.brand),
        "location": location_pts,
        "date": date_pts,
        "text": text_points(lost, found),
        "distance_km": round(distance, 2) if distance is not None else None,
    }
    total = sum(v for k, v in reasons.items() if k != "distance_km")
    return min(100, total), reasons


def run_matching(post):
    """
    Faol e'lonni qarama-qarshi turdagi faol e'lonlar bilan solishtiradi.
    Yangi mosliklar uchun ikkala egasiga bildirishnoma yuboradi.
    """
    config = SystemSettings.load()
    opposite = Post.Type.FOUND if post.type == Post.Type.LOST else Post.Type.LOST

    candidates = (
        Post.objects.filter(
            type=opposite,
            status=Post.Status.ACTIVE,
            category_id=post.category_id,
        )
        .exclude(author_id=post.author_id)
        .select_related("author", "place")
    )

    kept_ids = []
    created = []

    for other in candidates:
        lost, found = (post, other) if post.type == Post.Type.LOST else (other, post)
        result = compute_score(lost, found, config)
        if result is None:
            continue
        score, reasons = result
        if score < config.match_threshold:
            continue

        match, was_created = Match.objects.update_or_create(
            lost_post=lost,
            found_post=found,
            defaults={"score": score, "reasons": reasons},
        )
        kept_ids.append(match.pk)
        if was_created:
            created.append(match)
            notify(
                lost.author,
                Notification.Kind.MATCH_FOUND,
                "Mos buyum topildi",
                f"«{lost.title}» e'loningizga {score}% mos topilgan e'lon bor: «{found.title}».",
                post=found,
            )
            notify(
                found.author,
                Notification.Kind.MATCH_FOUND,
                "Egasi bo'lishi mumkin",
                f"«{found.title}» e'loningizga {score}% mos yo'qolgan e'lon bor: «{lost.title}».",
                post=lost,
            )

    # E'lon qayta faollashtirilganda eskirgan mosliklar tozalanadi
    Match.objects.filter(Q(lost_post=post) | Q(found_post=post)).exclude(
        pk__in=kept_ids
    ).delete()

    return created

            "reason", "reason_display", "text",
            "status", "status_display", "resolution_note", "created_at",

        else:

def _finish(report, moderator, status, note):
    report.status = status
    report.handled_by = moderator
    report.handled_at = timezone.now()
    report.resolution_note = note
    report.save()


@transaction.atomic
def resolve_report(report, moderator, note, reject_post_too=False):
    """Shikoyat o'rinli. Ixtiyoriy: e'lonni shu zahoti rad etish."""
    if report.status != Report.Status.OPEN:
        raise ValidationError("Bu shikoyat allaqachon ko'rib chiqilgan.")

    if reject_post_too:
        post = report.post
        if not post:
            raise ValidationError("Bu shikoyat e'longa emas, e'lonni rad etib bo'lmaydi.")
        if post.status not in (Post.Status.PENDING, Post.Status.ACTIVE):
            raise ValidationError(
                "E'lonni rad etib bo'lmaydi (u topshirish jarayonida yoki yopilgan)."
            )
        reject_post(post, moderator, note)

    _finish(report, moderator, Report.Status.RESOLVED, note)
    notify(
        report.reporter,
        Notification.Kind.REPORT_RESOLVED,
        "Shikoyatingiz ko'rib chiqildi",
        f"Shikoyatingiz o'rinli deb topildi. {note}",
        post=report.post,
    )
    return report


@transaction.atomic
def dismiss_report(report, moderator, note):
    """Shikoyat asossiz."""
    if report.status != Report.Status.OPEN:
        raise ValidationError("Bu shikoyat allaqachon ko'rib chiqilgan.")

    _finish(report, moderator, Report.Status.DISMISSED, note)
    notify(
        report.reporter,
        Notification.Kind.REPORT_RESOLVED,
        "Shikoyatingiz ko'rib chiqildi",
        f"Shikoyat asossiz deb topildi. {note}",
        post=report.post,
    )
    return report

@transaction.atomic
def reject_post(post, moderator, reason):
    from claims.models import Claim

    reason = (reason or "").strip()
    if not reason:
        raise ValidationError("Rad etish sababini yozing.")
    if post.status not in (Post.Status.PENDING, Post.Status.ACTIVE):
        raise ValidationError("Bu e'lonni rad etib bo'lmaydi.")

    now = timezone.now()
    post.status = Post.Status.REJECTED
    post.reject_reason = reason[:255]
    post.moderated_by = moderator
    post.moderated_at = now
    post.save(update_fields=["status", "reject_reason", "moderated_by", "moderated_at", "updated_at"])

    # Eskirgan mosliklar o'chiriladi
    Match.objects.filter(Q(lost_post=post) | Q(found_post=post)).delete()

    # E'longa kutilayotgan da'volar bekor qilinadi
    pending_claims = Claim.objects.filter(
        found_post=post, status=Claim.Status.PENDING
    ).select_related("claimant")
    for claim in pending_claims:
        claim.status = Claim.Status.CANCELLED
        claim.moderator = moderator
        claim.moderator_note = "E'lon rad etilgani sababli da'vo bekor qilindi."
        claim.reviewed_at = now
        claim.save()
        notify(
            claim.claimant,
            Notification.Kind.CLAIM_REJECTED,
            "Da'vo bekor qilindi",
            claim.moderator_note,
            post=post,
            claim=claim,
        )

    notify(
        post.author,
        Notification.Kind.POST_REJECTED,
        "E'loningiz rad etildi",
        f"«{post.title}»: {post.reject_reason}",
        post=post,
    )
    return post

    permission_classes = [IsAdministratorOrReadOnly]
    pagination_class = None

    def get_queryset(self):
        qs = Category.objects.annotate(

class PostViewSet(viewsets.ModelViewSet):
    serializer_class = PostSerializer
    filterset_class = PostFilter
    search_fields = ["title", "description", "brand", "location_detail"]
    ordering_fields = ["created_at", "event_date"]

    def get_permissions(self):
        if self.action in ("list", "retrieve"):
            return [AllowAny()]
        if self.action in ("update", "partial_update", "destroy"):
            return [IsAuthenticated(), IsAuthorOrAdmin()]
        return [IsAuthenticated()]

    def get_queryset(self):
        qs = Post.objects.select_related("category", "place", "author")
        if self.action == "list":
            return qs.filter(status=Post.Status.ACTIVE)  # umumiy ro'yxat: faqat faollar

        user = self.request.user
        if user.is_authenticated and user.is_moderator:
            return qs
        cond = Q(status=Post.Status.ACTIVE)
        if user.is_authenticated:
            cond |= Q(author=user)  # muallif o'z e'lonini har holatda ko'radi
        return qs.filter(cond)

    def perform_create(self, serializer):
        user = self.request.user
        config = SystemSettings.load()

        open_count = Post.objects.filter(
            author=user,
            status__in=[Post.Status.PENDING, Post.Status.ACTIVE, Post.Status.HANDOVER],
        ).count()
        if open_count >= config.max_active_posts_per_user:
            raise serializers.ValidationError(
                {"detail": f"Bir vaqtda {config.max_active_posts_per_user} tadan ko'p faol e'lon bo'lishi mumkin emas."}
            )

        status = Post.Status.PENDING if config.require_moderation else Post.Status.ACTIVE
        post = serializer.save(author=user, status=status)
        if post.status == Post.Status.ACTIVE:
            run_matching(post)

    def perform_update(self, serializer):
        post = serializer.instance
        if post.status in (Post.Status.HANDOVER, Post.Status.CLOSED):
            raise PermissionDenied("Bu e'lonni endi o'zgartirib bo'lmaydi.")

        # Tahrirdan keyin e'lon qayta tekshiruvga ketadi
        config = SystemSettings.load()
        status = Post.Status.PENDING if config.require_moderation else Post.Status.ACTIVE
        post = serializer.save(
            status=status, reject_reason="", moderated_by=None, moderated_at=None
        )
        if post.status == Post.Status.ACTIVE:
            run_matching(post)

    def perform_destroy(self, instance):
        user = self.request.user
        if instance.status in (Post.Status.HANDOVER, Post.Status.CLOSED) and not user.is_administrator:
            raise PermissionDenied("Topshirish jarayonidagi yoki yopilgan e'lonni o'chirib bo'lmaydi.")
        instance.delete()

    @action(detail=False, methods=["get"])
    def mine(self, request):
        """Mening e'lonlarim (hamma holatda)."""
        qs = Post.objects.filter(author=request.user).select_related("category", "place", "author")
        qs = self.filter_queryset(qs)
        page = self.paginate_queryset(qs)
        serializer = self.get_serializer(page, many=True)
        return self.get_paginated_response(serializer.data)

    @action(detail=True, methods=["get"])
    def matches(self, request, pk=None):
        """E'longa mos kelgan e'lonlar (faqat muallif va moderator)."""
        post = self.get_object()
        user = request.user
        if not (user.is_moderator or post.author_id == user.id):
            raise PermissionDenied("Mosliklarni faqat e'lon muallifi ko'ra oladi.")

        qs = Match.objects.filter(Q(lost_post=post) | Q(found_post=post)).select_related(
            "lost_post__category", "lost_post__place", "lost_post__author",
            "found_post__category", "found_post__place", "found_post__author",
        )
        items = []
        for m in qs:
            other = m.found_post if post.type == Post.Type.LOST else m.lost_post
            if other.status != Post.Status.ACTIVE and not user.is_moderator:
                continue
            items.append({
                "id": m.id,
                "score": m.score,
                "reasons": m.reasons,
                "post": PostSerializer(other, context=self.get_serializer_context()).data,
            })
        return Response({"count": len(items), "results": items})




class ModerationPostViewSet(
    mixins.ListModelMixin, mixins.RetrieveModelMixin, viewsets.GenericViewSet

        return qs

    @action(detail=True, methods=["post"])
    def approve(self, request, pk=None):
        post = approve_post(self.get_object(), request.user)
        return Response(self.get_serializer(post).data)

    @action(detail=True, methods=["post"])
    def reject(self, request, pk=None):
        serializer = RejectSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)
        post = reject_post(self.get_object(), request.user, serializer.validated_data["reason"])
        return Response(self.get_serializer(post).data)
